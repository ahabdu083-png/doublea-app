/**
 * ============================================================================
 * 2. RELIABLE VISUAL HELPERS & STRUCTURED MOCK COLLECTIONS
 * ============================================================================
 */
const SafeImage = ({ src, alt, className = '', style = {}, fallbackGradient }) => {
  const [failed, setFailed] = useState(false);

  useEffect(() => {
    setFailed(false);
  }, [src]);

  if (failed || !src) {
    return (
      <div
        className={`da-img-fallback ${className}`}
        style={{
          background: fallbackGradient || 'linear-gradient(135deg, #0755E8 0%, #061B55 60%, #18A0FF 100%)',
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'center',
          color: '#FFFFFF',
          fontWeight: 700,
          ...style
        }}
      >
        <Sparkles size={24} opacity={0.75} />
      </div>
    );
  }

  return (
    <img
      src={src}
      alt={alt}
      className={className}
      style={style}
      onError={() => setFailed(true)}
      loading="lazy"
    />
  );
};

                          onToggleBookmark={() => toggleBookmark(post.id)}
                          onAddComment={(txt) => handleAddComment(post.id, txt)}

                            className="da-wish-btn"

                            aria-label="Wishlist"
                          >

          <div
            className="da-moment-viewer"

              e.stopPropagation();

          >
            <SafeImage
              src={activeMoment.preview}
              alt={activeMoment.user}
              style={{ width: '100%', height: '100%', objectFit: 'cover' }}
            />
            <div className="da-moment-viewer-grad" />

                      key={m.label}

function QuickComposer({ currentUser, onQuickPublish, onOpenFullStudio }) {
  const [text, setText] = useState('');

  const submitQuick = (e) => {
    e.preventDefault();
    if (!text.trim()) return;
    onQuickPublish(text);
    setText('');
  };

  return (
    <form className="da-quick-composer" onSubmit={submitQuick}>
      <div style={{ display: 'flex', gap: 12, alignItems: 'center' }}>
        <SafeImage
          src={currentUser.avatar}
          alt={currentUser.name}
          style={{ width: 44, height: 44, borderRadius: 15, objectFit: 'cover', flexShrink: 0 }}
        />
        <input
          type="text"
          className="da-quick-input"
          placeholder={`Broadcast a spark, ${currentUser.name.split(' ')[0]}...`}
          value={text}
          onChange={(e) => setText(e.target.value)}
        />
      </div>

      <div className="da-quick-actions">
        <div style={{ display: 'flex', gap: 6, flexWrap: 'wrap' }}>
          <button type="button" className="da-mini-tool" onClick={onOpenFullStudio}>
            <ImageIcon size={16} color="#0755E8" /> Photo
          </button>
          <button type="button" className="da-mini-tool" onClick={onOpenFullStudio}>
            <Video size={16} color="#18A0FF" /> Video
          </button>
          <button type="button" className="da-mini-tool" onClick={onOpenFullStudio}>
            <Smile size={16} color="#061B55" /> Feeling
          </button>
        </div>
        <button type="submit" className="da-btn-primary" style={{ padding: '8px 16px', fontSize: 13 }}>
          Spark <Send size={14} />
        </button>
      </div>
    </form>
  );
}

function PostCard({ post, isOwnPost, onToggleLike, onToggleBookmark, onAddComment, onShare, onDelete }) {
  const [commentsOpen, setCommentsOpen] = useState(false);
  const [commentText, setCommentText] = useState('');

  const submitReply = (e) => {
    e.preventDefault();
    if (!commentText.trim()) return;
    onAddComment(commentText);
    setCommentText('');
    setCommentsOpen(true);
  };

  return (
    <article className="da-post-card">
      <div className="da-post-head">
        <SafeImage
          src={post.authorAvatar}
          alt={post.authorName}
          style={{ width: 46, height: 46, borderRadius: 16, objectFit: 'cover', flexShrink: 0 }}
        />
        <div style={{ flex: 1, minWidth: 0 }}>
          <div style={{ display: 'flex', alignItems: 'center', gap: 6, flexWrap: 'wrap' }}>
            <span style={{ fontWeight: 800, fontSize: 15, color: '#061B55' }}>{post.authorName}</span>
            {post.verified && <ShieldCheck size={15} color="#0755E8" />}
            <span style={{ fontSize: 13, color: '#4A5B82' }}>{post.authorHandle}</span>
            {post.feeling && <span className="da-feeling-chip">feeling {post.feeling}</span>}
          </div>
          <div className="da-truncate" style={{ fontSize: 12, color: '#4A5B82', marginTop: 2 }}>
            {post.authorRole} • {post.timestamp}
          </div>
        </div>
        <span className="da-post-cat-tag da-hide-sm">{post.category}</span>
      </div>

      <p className="da-post-body">{post.content}</p>

      {post.tags && post.tags.length > 0 && (
        <div className="da-post-tags">
          {post.tags.map((tag, idx) => (
            <span key={idx}>{tag}</span>
          ))}
        </div>
      )}

      {post.mediaUrl && (
        <div className="da-post-media">
          <SafeImage
            src={post.mediaUrl}
            alt="Post visual"
            style={{ width: '100%', maxHeight: 390, objectFit: 'cover', borderRadius: 20 }}
          />
          {post.mediaType === 'video' && (
            <div className="da-video-overlay-pill">
              <Play size={16} fill="#FFFFFF" color="#FFFFFF" />
              <span>Spatial Clip • {post.videoDuration || '01:00'}</span>
            </div>
          )}
        </div>
      )}

      <div className="da-post-footer">
        <div style={{ display: 'flex', gap: 8 }}>
          <button
            className={`da-action-chip ${post.liked ? 'liked' : ''}`}
            onClick={onToggleLike}
          >
            <Heart size={17} fill={post.liked ? '#0755E8' : 'none'} color={post.liked ? '#0755E8' : '#4A5B82'} />
            <span>{post.likes}</span>
          </button>

          <button
            className={`da-action-chip ${commentsOpen ? 'active' : ''}`}
            onClick={() => setCommentsOpen(!commentsOpen)}
          >
            <MessageCircle size={17} color={commentsOpen ? '#0755E8' : '#4A5B82'} />
            <span>{post.comments.length}</span>
          </button>

          <button className="da-action-chip" onClick={onShare}>
            <Share2 size={17} color="#4A5B82" />
            <span>{post.shares}</span>
          </button>
        </div>

        <div style={{ display: 'flex', gap: 6 }}>
          <button
            className={`da-icon-btn ${post.bookmarked ? 'bookmarked' : ''}`}
            onClick={onToggleBookmark}
            aria-label="Bookmark"
          >
            <Bookmark size={18} fill={post.bookmarked ? '#0755E8' : 'none'} color={post.bookmarked ? '#0755E8' : '#4A5B82'} />
          </button>
          {isOwnPost && onDelete && (
            <button
              className="da-icon-btn"
              onClick={onDelete}
              aria-label="Delete Spark"
              title="Delete Spark"
            >
              <Trash2 size={17} color="#4A5B82" />
            </button>
          )}
        </div>
      </div>

      {commentsOpen && (
        <div className="da-thread-drawer">
          {post.comments.map((c) => (
            <div key={c.id} className="da-comment-row">
              <SafeImage
                src={c.avatar}
                alt={c.user}
                style={{ width: 32, height: 32, borderRadius: 10, objectFit: 'cover', flexShrink: 0 }}
              />
              <div className="da-comment-bubble">
                <div style={{ display: 'flex', justifyContent: 'space-between', fontSize: 12 }}>
                  <strong style={{ color: '#061B55' }}>{c.user}</strong>
                  <span style={{ color: '#4A5B82' }}>{c.time}</span>
                </div>
                <p style={{ fontSize: 13, color: '#061B55', marginTop: 3 }}>{c.text}</p>
              </div>
            </div>
          ))}

          <form onSubmit={submitReply} className="da-reply-form">
            <input
              type="text"
              placeholder="Add a thoughtful reply to this thread..."
              value={commentText}
              onChange={(e) => setCommentText(e.target.value)}
            />
            <button type="submit" className="da-btn-primary" style={{ padding: '8px 14px', fontSize: 12 }}>
              Reply
            </button>
          </form>
        </div>
      )}
    </article>
  );
}
